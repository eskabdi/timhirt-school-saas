# Scope: every pg_policy on public+storage (name, cmd, permissive, roles, md5(qual#with_check)),

#        every non-extension public function (signature, prosecdef, proconfig, md5(prosrc)).
# Not in scope: table/function ACLs (the harness shim lacks Supabase default grants, L-08 -> WP-01).

## Query
select 'policy|'||n.nspname||'.'||c.relname||'|'||p.polname||'|'||p.polcmd::text||'|'||p.polpermissive::text||'|'||coalesce((select string_agg(case when r=0 then 'public' else r::regrole::text end, ',' order by 1) from unnest(p.polroles) r),'')||'|'||md5(coalesce(pg_get_expr(p.polqual,p.polrelid),'')||'#'||coalesce(pg_get_expr(p.polwithcheck,p.polrelid),'')) as l
from pg_policy p join pg_class c on c.oid=p.polrelid join pg_namespace n on n.oid=c.relnamespace where n.nspname in ('public','storage')
union all
select 'rls|public.'||c.relname||'|'||c.relrowsecurity::text||'|'||c.relforcerowsecurity::text from pg_class c join pg_namespace n on n.oid=c.relnamespace where n.nspname='public' and c.relkind in ('r','p')
union all
select 'fn|'||p.oid::regprocedure::text||'|'||p.prosecdef::text||'|'||coalesce(array_to_string(p.proconfig,','),'')||'|'||md5(p.prosrc) from pg_proc p join pg_namespace n on n.oid=p.pronamespace where n.nspname='public' and not exists (select 1 from pg_depend d where d.classid='pg_proc'::regclass and d.objid=p.oid and d.deptype='e')
order by 1;
