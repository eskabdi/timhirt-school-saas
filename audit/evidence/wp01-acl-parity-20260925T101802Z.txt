# R6 WP-01 (L-08): effective privileges, harness (shim with Supabase default grants) vs production
# Captured 20260925T101802Z. Production via Management API /database/query read_only=true (livqynxlibmccaycseer, 108 migrations);
# harness = supabase/tests/run.sh DB (108 migrations) with the WP-01 shim.
# Scope: every non-extension function in public (secdef; EXECUTE for anon/authenticated/service_role)
#        and every table/view in public (SELECT for anon/authenticated, INSERT for authenticated).

## Query
select p.oid::regprocedure::text||'|'||p.prosecdef::text||'|'||has_function_privilege('anon',p.oid,'execute')::text||'|'||has_function_privilege('authenticated',p.oid,'execute')::text||'|'||has_function_privilege('service_role',p.oid,'execute')::text as l from pg_proc p join pg_namespace n on n.oid=p.pronamespace where n.nspname='public' and not exists (select 1 from pg_depend d where d.classid='pg_proc'::regclass and d.objid=p.oid and d.deptype='e') union all select 'table|'||c.relname||'|'||has_table_privilege('anon',c.oid,'select')::text||'|'||has_table_privilege('authenticated',c.oid,'select')::text||'|'||has_table_privilege('authenticated',c.oid,'insert')::text from pg_class c join pg_namespace n on n.oid=c.relnamespace where n.nspname='public' and c.relkind in ('r','p','v') and not exists (select 1 from pg_depend d where d.classid='pg_class'::regclass and d.objid=c.oid and d.deptype='e') order by 1
